Local Multi-Cloud (AWS + Azure) Execution Quick-Start

# 1. Authenticate to Wiz, HCP Terraform, AWS (CLI/SSO), and Azure (az login)
export WIZ_API_URL="https://api.us1.app.wiz.io/graphql"
export WIZ_CLIENT_ID="your-wiz-service-account-client-id"
export WIZ_CLIENT_SECRET="your-wiz-service-account-secret"
export TFC_TOKEN="your-hcp-terraform-or-tfe-api-token"
export TFC_ORG="your-financial-org"
aws sso login && az login

# 2. Run Automated Daily Multi-Cloud Policy Sync across AWS, Azure, Wiz, HCP Terraform & Checkov
python3 azure_to_wiz_mapper.py \
  --sync-all-daily \
  --live-aws --aws-region us-east-1 \
  --live-azure --management-group-id "mg-financial-enterprise" \
  --live-tfc --tfc-org "$TFC_ORG" \
  --live-wiz --cloud-providers AZURE AWS \
  --baseline-audit-json ./data/baseline_sdlc_audit.json \
  --output-csv ./reports/sdlc_cloud_policy_governance_audit.csv \
  --output-json ./data/baseline_sdlc_audit.json \
  --output-html ./reports/interactive_mapping_report.html
